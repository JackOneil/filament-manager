function _auditEscHtml(s) {
    return s.replace(/&/g, '&').replace(//g, '>');
}

function _auditPrettyJson(raw) {
    if (!raw) return '';
    try { return JSON.stringify(JSON.parse(raw), null, 2); }
    catch (e) { return raw; }
}

// Myers-style LCS diff on arrays of strings
function _auditLcsDiff(before, after) {
    const m = before.length, n = after.length;
    // Build LCS DP table (only 2 rows needed)
    const dp = Array.from({ length: m + 1 }, () => new Int32Array(n + 1));
    for (let i = 1; i <= m; i++)
        for (let j = 1; j <= n; j++)
            dp[i][j] = before[i-1] === after[j-1] ? dp[i-1][j-1] + 1 : Math.max(dp[i-1][j], dp[i][j-1]);
    // Backtrack
    const ops = [];
    let i = m, j = n;
    while (i > 0 || j > 0) {
        if (i > 0 && j > 0 && before[i-1] === after[j-1]) {
            ops.unshift({ t: '=', v: before[i-1] }); i--; j--;
        } else if (j > 0 && (i === 0 || dp[i][j-1] >= dp[i-1][j])) {
            ops.unshift({ t: '+', v: after[j-1] }); j--;
        } else {
            ops.unshift({ t: '-', v: before[i-1] }); i--;
        }
    }
    return ops;
}

function auditDiffPanel(el) {
    return {
        view: 'diff',
        beforeText: _auditPrettyJson(el.dataset.before || ''),
        afterText:  _auditPrettyJson(el.dataset.after  || ''),
        get diffHtml() {
            const b = this.beforeText, a = this.afterText;
            if (!b && !a) return '-';
            const bLines = b ? b.split('\n') : [];
            const aLines = a ? a.split('\n') : [];
            if (!b) return aLines.map(l =>
                `
+${_auditEscHtml(l)}
`
            ).join('');
            if (!a) return bLines.map(l =>
                `
-${_auditEscHtml(l)}
`
            ).join('');
            const ops = _auditLcsDiff(bLines, aLines);
            // Count changes for context collapsing
            let html = '';
            for (let idx = 0; idx < ops.length; idx++) {
                const op = ops[idx];
                const esc = _auditEscHtml(op.v);
                if (op.t === '=') {
                    html += `
 ${esc}
`;
                } else if (op.t === '-') {
                    html += `
-${esc}
`;
                } else {
                    html += `
+${esc}
`;
                }
            }
            return html || '' + (window.__i18n && window.__i18n[window.__helpLang||'cs'] && window.__i18n[window.__helpLang||'cs']['audit_no_changes'] || '{{ t('audit_no_changes')|e }}') + '';
        }
    };
}
